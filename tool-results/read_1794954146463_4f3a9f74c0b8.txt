    25→    25→// Characters from RTL scripts (Hebrew, Arabic, Syriac, Thaana, …) — used by
    26→    26→// the "auto" text-direction rule (first strong character, like dir=auto).
    27→    27→const RTL_CHAR_RE = /[\u0590-\u05FF\u0600-\u06FF\u0750-\u077F\u08A0-\u08FF\uFB1D-\uFDFF\uFE70-\uFEFF]/;
    28→    28→
    29→    29→// Resolve the base text direction for rendering. "auto" follows the standard
    30→    30→// first-strong rule: scan for the first character with a strong direction —
    31→    31→// an RTL-script letter wins, otherwise the first regular letter means LTR
    32→    32→// (digits and punctuation are neutral and skipped). Export and preview use
    33→    33→// the same rule so both always agree.
    34→    34→export function resolveTextDirection(
    35→    35→  text: string,
    36→    36→  setting: TextDirection
    37→    37→): "ltr" | "rtl" {
    38→    38→  if (setting !== "auto") return setting;
    39→    39→  for (const ch of text) {
    40→    40→    if (RTL_CHAR_RE.test(ch)) return "rtl";
    41→    41→    if (/\p{L}/u.test(ch)) return "ltr";
    42→    42→  }
    43→    43→  return "ltr";
    44→    44→}

   129→   129→export function hexToRgba(hex: string, alpha: number): string {
   130→   130→  let h = hex.replace("#", "");
   131→   131→  if (h.length === 3) {
   132→   132→    h = h
   133→   133→      .split("")
   134→   134→      .map((c) => c + c)
   135→   135→      .join("");
   136→   136→  }
   137→   137→  const r = parseInt(h.slice(0, 2), 16) || 0;
   138→   138→  const g = parseInt(h.slice(2, 4), 16) || 0;
   139→   139→  const b = parseInt(h.slice(4, 6), 16) || 0;
   140→   140→  return `rgba(${r},${g},${b},${alpha})`;
   141→   141→}

   145→   145→export function contrastFor(hex: string): string {
   146→   146→  let h = hex.replace("#", "");
   147→   147→  if (h.length === 3) {
   148→   148→    h = h
   149→   149→      .split("")
   150→   150→      .map((c) => c + c)
   151→   151→      .join("");
   152→   152→  }
   153→   153→  const r = parseInt(h.slice(0, 2), 16) || 0;
   154→   154→  const g = parseInt(h.slice(2, 4), 16) || 0;
   155→   155→  const b = parseInt(h.slice(4, 6), 16) || 0;
   156→   156→  const lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   157→   157→  return lum > 0.6 ? "#000000" : "#ffffff";
   158→   158→}

   206→   206→export function computeTextBlock(
   207→   207→  height: number,
   208→   208→  fontSize: number,
   209→   209→  lineHeight: number,
   210→   210→  lineCount: number,
   211→   211→  vAlign: VAlign
   212→   212→): TextBlock {
   213→   213→  const h = Math.max(1, lineCount) * fontSize * lineHeight;
   214→   214→  const margin = 24;
   215→   215→  if (vAlign === "top") return { top: margin, bottom: margin + h, height: h };
   216→   216→  if (vAlign === "bottom")
   217→   217→    return { top: height - margin - h, bottom: height - margin, height: h };
   218→   218→  const top = (height - h) / 2;
   219→   219→  return { top, bottom: top + h, height: h };
   220→   220→}

   263→   263→function cssCodeText(o: OverlayConfig): string {
   264→   264→  return `font-variation-settings: ${buildCssFontVariationSettings(

   269→   269→// The widest the CSS code string can get for the current axes. The rendered
   270→   270→// value strings vary in length from frame to frame — measuring against the
   271→   271→// widest variant keeps the CSS line's font size and bar geometry CONSTANT
   272→   272→// across frames (otherwise the bar visibly pulses while the animation runs).
   273→   273→function cssCodeWidestText(o: OverlayConfig): string {
   274→   274→  const dec = o.valueDecimals;
   275→   275→  const parts = o.axes.map((a) => {
   276→   276→    const widest = [a.min, a.max]
   277→   277→      .map((v) => formatValue(v, dec))
   278→   278→      .reduce((p, c) => (c.length > p.length ? c : p), "");
   279→   279→    return `"${a.tag}" ${widest}`;
   280→   280→  });
   281→   281→  return `font-variation-settings: ${parts.join(", ")};`;
   282→   282→}

   284→   284→function minMaxChars(o: OverlayConfig): number {
   285→   285→  let chars = 0;
   286→   286→  for (const a of o.axes) {
   287→   287→    const len = Math.max(fmtNum(a.min).length, fmtNum(a.max).length);
   288→   288→    if (len > chars) chars = len;
   289→   289→  }
   290→   290→  return Math.max(chars, 3);
   291→   291→}

   293→   293→// Shared geometry for the overlay — used by both the SVG frame renderer and
   294→   294→// the DOM live preview so both look identical. All values are in canvas px.
   295→   295→export function computeOverlayMetrics(
   296→   296→  width: number,
   297→   297→  height: number,
   298→   298→  o: OverlayConfig,
   299→   299→  textBlock: TextBlock
   300→   300→): OverlayMetrics {
   301→   301→  const s = o.scale;
   302→   302→  // The slider block only exists when the "slider" toggle is on; the name,
   303→   303→  // min–max labels and the value pill are decorations that live inside it.
   304→   304→  const slidersOn = o.showSliders && o.axes.length > 0;
   305→   305→  const fontPx = Math.max(9, Math.round(width * 0.016 * s));
   306→   306→  const pad = Math.max(8, Math.round(width * 0.014 * s));
   307→   307→  const innerPad = Math.max(6, Math.round(pad * 0.85));
   308→   308→  const panelRx = Math.max(6, Math.round(innerPad * 0.9));
   309→   309→  const trackH = Math.max(3, Math.round(fontPx * 0.3));
   310→   310→  const thumbR = Math.max(
   311→   311→    3,
   312→   312→    Math.round(fontPx * 0.48 * Math.min(3, Math.max(0.4, o.thumbScale)))
   313→   313→  );
   314→   314→  const thumbD = thumbR * 2;
   315→   315→  const minMaxFont = Math.max(8, Math.round(fontPx * 0.78));
   316→   316→
   317→   317→  // Inline axis name — drawn BEFORE the track on the same row, so toggling
   318→   318→  // the name never changes the height of the slider stack.
   319→   319→  const hasName = slidersOn && (o.showName || o.showTag) && o.axes.length > 0;
   320→   320→  let nameChars = 0;
   321→   321→  if (hasName) {
   322→   322→    for (const a of o.axes) {
   323→   323→      let len = 0;
   324→   324→      if (o.showName) len += a.name.length;
   325→   325→      if (o.showTag) len += a.tag.length + 2;
   326→   326→      if (len > nameChars) nameChars = len;
   327→   327→    }
   328→   328→  }
   329→   329→  const nameW = hasName ? Math.ceil(nameChars * 0.62 * fontPx) : 0;
   330→   330→  // Horizontal gap between the inline name and the min label / track.
   331→   331→  const nameGapPx = Math.round(
   332→   332→    Math.min(Math.max(o.nameGap, 0), 4) * fontPx
   333→   333→  );
   334→   334→  const textGapPx = Math.round(
   335→   335→    Math.min(Math.max(o.textGap, 0), 6) * fontPx
   336→   336→  );
   337→   337→
   338→   338→  // Value pill riding ON the thumb (Material-style label). It replaces the
   339→   339→  // thumb and adds NO extra row, so the stack height stays identical whether
   340→   340→  // the value number is on or off. Width uses the widest value string the
   341→   341→  // axis can ever show (min/max at the chosen decimals) so the pill geometry
   342→   342→  // is frame-stable — no pulsing mid-animation.
   343→   343→  const pillOn = slidersOn && o.showLiveValue;
   344→   344→  const pillFont = Math.max(8, Math.round(fontPx * 0.8));
   345→   345→  const pillPadX = Math.max(6, Math.round(pillFont * 0.55));
   346→   346→  const pillPadY = Math.max(3, Math.round(pillFont * 0.3));
   347→   347→  let valueChars = 3;
   348→   348→  for (const a of o.axes) {
   349→   349→    const len = Math.max(
   350→   350→      formatValue(a.min, o.valueDecimals).length,
   351→   351→      formatValue(a.max, o.valueDecimals).length
   352→   352→    );
   353→   353→    if (len > valueChars) valueChars = len;
   354→   354→  }
   355→   355→  const pillTextW = Math.ceil(valueChars * 0.62 * pillFont);
   356→   356→  const pillH = Math.max(thumbD, Math.round(pillFont * 1.5) + pillPadY * 2);
   357→   357→  const pillW = Math.max(
   358→   358→    thumbD + Math.round(thumbD * 0.4),
   359→   359→    pillTextW + pillPadX * 2
   360→   360→  );
   361→   361→  const pillTextFill = contrastFor(o.color);
   362→   362→  // The value pill rides ON the thumb and its centre follows the value across
   363→   363→  // the FULL track (no end-clamping), so it overhangs the track ends by up to
   364→   364→  // half its width. Reserve that much room on both ends of the row so the
   365→   365→  // pill always stays inside the panel while the thumb keeps moving.
   366→   366→  const pillBuffer = pillOn ? Math.ceil(pillW / 2) : 0;
   367→   367→
   368→   368→  const minMaxLabelW =
   369→   369→    slidersOn && o.showMinMax
   370→   370→      ? Math.ceil(minMaxChars(o) * 0.62 * minMaxFont)
   371→   371→      : 0;
   372→   372→  const flankGap = Math.max(4, Math.round(fontPx * 0.5));
   373→   373→  const trackRowH = Math.max(
   374→   374→    Math.round(thumbD * 1.25),
   375→   375→    Math.round(minMaxFont * 1.35)
   376→   376→  );
   377→   377→
   378→   378→  const nameSpan = nameW + (hasName ? nameGapPx : 0);
   379→   379→  const trackFlank = minMaxLabelW > 0 ? minMaxLabelW * 2 + flankGap * 2 : 0;
   380→   380→  const availForTrack =
   381→   381→    width - pad * 2 - innerPad * 2 - nameSpan - trackFlank - pillBuffer * 2;
   382→   382→  const trackW = Math.max(
   383→   383→    fontPx * 6,
   384→   384→    Math.min(
   385→   385→      Math.round(width * 0.3 * s),
   386→   386→      Math.max(availForTrack, fontPx * 6)
   387→   387→    )
   388→   388→  );
   389→   389→
   390→   390→  const trackRowW = pillBuffer * 2 + nameSpan + trackFlank + trackW;
   391→   391→  const maxBlock = Math.max(fontPx * 6, width - pad * 2 - innerPad * 2);
   392→   392→  const blockW = Math.min(maxBlock, trackRowW);
   393→   393→  const panelW = blockW + innerPad * 2;
   394→   394→
   395→   395→  const axisGap = Math.round(fontPx * 1.2);
   396→   396→  const rowStack = trackRowH; // one compact row per axis
   397→   397→  const n = o.axes.length;
   398→   398→  const panelH =
   399→   399→    slidersOn && n > 0 ? innerPad * 2 + rowStack * n + axisGap * (n - 1) : 0;
   400→   400→
   401→   401→  // CSS code line — auto-shrink font if the string is longer than the bar.
   402→   402→  // The width estimate uses the widest possible value strings so the font
   403→   403→  // size (and therefore the bar geometry) stays constant across frames.
   404→   404→  const cssText = cssCodeText(o);
   405→   405→  const cssWidest = cssCodeWidestText(o);
   406→   406→  const availCss = Math.max(40, width - pad * 2 - innerPad * 2);
   407→   407→  const cssEst = 0.62 * fontPx * Math.max(1, cssWidest.length);
   408→   408→  const cssFontPx =
   409→   409→    cssEst > availCss
   410→   410→      ? Math.max(
   411→   411→          8,
   412→   412→          Math.floor(availCss / (0.62 * Math.max(1, cssWidest.length)))
   413→   413→        )
   414→   414→      : fontPx;
   415→   415→  const cssTextW = Math.ceil(cssWidest.length * 0.62 * cssFontPx);
   416→   416→  const cssBarH = o.showCss ? Math.round(cssFontPx * 1.5 + innerPad * 1.1) : 0;
   417→   417→  const cssGapPx = Math.round(
   418→   418→    Math.min(Math.max(o.cssGap, 0), 4) * fontPx
   419→   419→  );
   420→   420→  const cssOn = o.showCss && cssBarH > 0;
   421→   421→  const cssBarW = cssOn
   422→   422→    ? Math.min(
   423→   423→        width - pad * 2,
   424→   424→        o.centered ? cssTextW + innerPad * 2 : width - pad * 2
   425→   425→      )
   426→   426→    : 0;
   427→   427→
   428→   428→  // ── Vertical placement ────────────────────────────────────────────────
   429→   429→  // The whole stack (slider panel + CSS line) hugs the text block with
   430→   430→  // textGapPx between them, clamped so it never leaves the canvas.
   431→   431→  const stackH = panelH + (slidersOn && cssOn ? cssGapPx + cssBarH : cssOn ? cssBarH : 0);
   432→   432→  const minTop = pad;

   467→   467→

   510→   510→// Thumb marker in every supported shape, drawn centered on (cx, cy).
   511→   511→function thumbSvg(
   512→   512→  shape: ThumbShape,
   513→   513→  cx: number,
   514→   514→  cy: number,
   515→   515→  r: number,
   516→   516→  fill: string

   533→   533→function buildOverlaySvg(
   534→   534→  o: OverlayConfig,
   535→   535→  width: number,
   536→   536→  height: number,
   537→   537→  textBlock: TextBlock
   538→   538→): string {
   539→   539→  const m = computeOverlayMetrics(width, height, o, textBlock);
   540→   540→  const parts: string[] = [];
   541→   541→  const fontAttr = `font-family="${escapeAttr(o.font)}"`;
   542→   542→  const fill = escapeAttr(o.color);
   543→   543→  if (o.showSliders && m.panelH > 0) {
   544→   544→    if (o.bgOpacity > 0) {
   545→   545→      parts.push(
   546→   546→        `<rect x="${m.panelX}" y="${m.panelY}" width="${m.panelW}" height="${m.panelH}" rx="${m.panelRx}" fill="${escapeAttr(o.bgColor)}" fill-opacity="${round2(o.bgOpacity)}"/>`
   547→   547→      );
   548→   548→    }
   549→   549→    const contentX = m.panelX + m.innerPad;
   550→   550→
   551→   551→    o.axes.forEach((a, i) => {
   552→   552→      const y = m.panelY + m.innerPad + i * (m.trackRowH + m.axisGap);
   553→   553→      const cy = y + m.trackRowH / 2;
   554→   554→      const t = clamp01((a.value - a.min) / (a.max - a.min || 1));
   555→   555→      const trackX = contentX + m.trackLeft;
   556→   556→      const fillW = t * m.trackW;
   557→   557→      const thumbCx = trackX + fillW;
   558→   558→
   559→   559→      // Inline axis name, BEFORE the track (same row → constant stack height).
   560→   560→      // Right-aligned inside its fixed-width slot, so every name ends at the
   561→   561→      // SAME distance from the start of its own track.
   562→   562→      if (m.nameW > 0) {
   563→   563→        const label = [
   564→   564→          o.showName ? escapeXml(a.name) : "",
   565→   565→          o.showTag ? `(${escapeXml(a.tag)})` : "",
   566→   566→        ]
   567→   567→          .filter(Boolean)
   568→   568→          .join(" ");
   569→   569→        parts.push(
   570→   570→          `<text x="${round2(contentX + m.pillBuffer + m.nameW)}" y="${round2(cy + m.fontPx * 0.35)}" text-anchor="end" fill="${fill}" ${fontAttr} font-size="${m.fontPx}">${label}</text>`
   571→   571→        );
   572→   572→      }
   573→   573→
   574→   574→      // Min–max labels flanking the track.
   575→   575→      if (m.minMaxLabelW > 0) {
   576→   576→        const lb = cy + m.minMaxFont * 0.35;
   577→   577→        parts.push(
   578→   578→          `<text x="${round2(trackX - m.flankGap)}" y="${round2(lb)}" text-anchor="end" fill="${fill}" fill-opacity="0.75" ${fontAttr} font-size="${m.minMaxFont}">${fmtNum(a.min)}</text>`
   579→   579→        );
   580→   580→        parts.push(
   581→   581→          `<text x="${round2(trackX + m.trackW + m.flankGap)}" y="${round2(lb)}" text-anchor="start" fill="${fill}" fill-opacity="0.75" ${fontAttr} font-size="${m.minMaxFont}">${fmtNum(a.max)}</text>`
   582→   582→        );
   583→   583→      }
   584→   584→
   585→   585→      parts.push(
   586→   586→        `<rect x="${trackX}" y="${round2(cy - m.trackH / 2)}" width="${m.trackW}" height="${m.trackH}" rx="${m.trackH / 2}" fill="${fill}" fill-opacity="0.22"/>`
   587→   587→      );
   588→   588→      if (fillW > 0) {
   589→   589→        parts.push(
   590→   590→          `<rect x="${trackX}" y="${round2(cy - m.trackH / 2)}" width="${round2(fillW)}" height="${m.trackH}" rx="${m.trackH / 2}" fill="${fill}" fill-opacity="0.7"/>`
   591→   591→        );
   592→   592→      }
   593→   593→
   594→   594→      // Value pill riding ON the thumb (replaces it). The pill centre follows
   595→   595→      // the value across the FULL track — no end-clamping — so the thumb is
   596→   596→      // always moving whenever the number changes. It may overhang the track
   597→   597→      // ends; the row's pillBuffer reserves room for that inside the panel.
   598→   598→      // The centre is snapped to whole canvas pixels so it rasterises
   599→   599→      // identically frame-to-frame instead of shimmering sub-pixel.
   600→   600→      if (m.pillOn) {
   601→   601→        const pcx = Math.round(thumbCx);
   602→   602→        const prx = pillRadius(o.thumbShape, Math.min(m.pillW, m.pillH));
   603→   603→        parts.push(
   604→   604→          `<rect x="${round2(pcx - m.pillW / 2)}" y="${round2(cy - m.pillH / 2)}" width="${round2(m.pillW)}" height="${round2(m.pillH)}" rx="${round2(prx)}" fill="${fill}"/>`
   605→   605→        );
   606→   606→        parts.push(
   607→   607→          `<text x="${pcx}" y="${round2(cy + m.pillFont * 0.35)}" text-anchor="middle" fill="${m.pillTextFill}" ${fontAttr} font-size="${m.pillFont}">${formatValue(a.value, o.valueDecimals)}</text>`
   608→   608→        );
   609→   609→      } else {
   610→   610→        parts.push(thumbSvg(o.thumbShape, thumbCx, cy, m.thumbR, fill));
   611→   611→      }
   612→   612→    });
   613→   613→  }
   614→   614→
   615→   615→  if (m.cssOn) {
   616→   616→    const cssText = cssCodeText(o);
   617→   617→    if (o.bgOpacity > 0) {
   618→   618→      parts.push(
   619→   619→        `<rect x="${m.cssBarX}" y="${m.cssBarY}" width="${m.cssBarW}" height="${m.cssBarH}" rx="${m.panelRx}" fill="${escapeAttr(o.bgColor)}" fill-opacity="${round2(o.bgOpacity)}"/>`
   620→   620→      );
   621→   621→    }
   622→   622→    const anchor = o.centered ? "middle" : "start";
   623→   623→    const tx = o.centered
   624→   624→      ? m.cssBarX + m.cssBarW / 2
   625→   625→      : m.cssBarX + m.innerPad;
   626→   626→    parts.push(
   627→   627→      `<text x="${round2(tx)}" y="${round2(m.cssBarY + m.cssBarH / 2 + m.cssFontPx * 0.36)}" text-anchor="${anchor}" fill="${fill}" ${fontAttr} font-size="${m.cssFontPx}" xml:space="preserve">${escapeXml(cssText)}</text>`
   628→   628→    );
   629→   629→  }
   630→   630→
   631→   631→  return parts.join("");

   634→   634→export function buildSvgString(cfg: RenderConfig): string {
   635→   635→  const {
   636→   636→    text,
   637→   637→    fontBase64,
   638→   638→    fontCssFormat,
   639→   639→    fontMime,
   640→   640→    variation,
   641→   641→    fontSize,
   642→   642→    textColor,
   643→   643→    bgColor,
   644→   644→    width,
   645→   645→    height,
   646→   646→    letterSpacing,
   647→   647→    lineHeight,
   648→   648→    align,
   649→   649→    vAlign,
   650→   650→    direction,
   651→   651→    overlay,
   652→   652→  } = cfg;
   653→   653→
   654→   654→  const lines = text.length ? text.split("\n") : [" "];
   655→   655→  const textBlock = computeTextBlock(
   656→   656→    height,
   657→   657→    fontSize,
   658→   658→    lineHeight,
   659→   659→    lines.length,
   660→   660→    vAlign
   661→   661→  );
   662→   662→  const top = textBlock.top;
   663→   663→  // Horizontal alignment is PHYSICAL (left/center/right stay where they are
   664→   664→  // on screen regardless of text direction). SVG text-anchor is LOGICAL —
   665→   665→  // with an RTL base direction start/end swap sides — so map the physical
   666→   666→  // alignment onto the matching logical anchor. anchorX stays physical.
   667→   667→  const anchorX =
   668→   668→    align === "center" ? width / 2 : align === "right" ? width - 24 : 24;
   669→   669→  const physAnchor =
   670→   670→    align === "center" ? "middle" : align === "right" ? "end" : "start";
   671→   671→  const anchor =
   672→   672→    direction === "rtl" && physAnchor === "start"
   673→   673→      ? "end"
   674→   674→      : direction === "rtl" && physAnchor === "end"
   675→   675→        ? "start"
   676→   676→        : physAnchor;
   677→   677→  // Establish the RTL base direction on the text element. Per SVG 1.1 the
   678→   678→  // direction property needs unicode-bidi to open an embedding. LTR needs no
   679→   679→  // style — it is the SVG default.
   680→   680→  const dirStyle =
   681→   681→    direction === "rtl"
   682→   682→      ? " direction: rtl; unicode-bidi: embed;"
   683→   683→      : "";
   684→   684→
   685→   685→  const tspans = lines
   686→   686→    .map((line, i) => {
   687→   687→      const y = top + i * fontSize * lineHeight + fontSize * 0.78;
   688→   688→      return `<tspan x="${anchorX}" y="${round2(y)}">${escapeXml(line) || "&#8203;"}</tspan>`;
   689→   689→    })
   690→   690→    .join("");
   691→   691→
   692→   692→  const variationCss = buildVariationString(variation);
   693→   693→  const letterSpacingCss =
   694→   694→    letterSpacing !== 0 ? `letter-spacing: ${letterSpacing}px;` : "";
   695→   695→
   696→   696→  const overlaySvg =
   697→   697→    overlay &&
   698→   698→    ((overlay.showSliders && overlay.axes.length > 0) || overlay.showCss)
   699→   699→      ? buildOverlaySvg(overlay, width, height, textBlock)
   700→   700→      : "";
   701→   701→
   702→   702→  return `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}" viewBox="0 0 ${width} ${height}">
   703→   703→<defs>
   704→   704→<style>
   705→   705→@font-face { font-family: 'VFStudio'; src: url(data:${escapeAttr(fontMime)};base64,${fontBase64}) format('${escapeAttr(fontCssFormat)}'); }
   706→   706→.vftxt { font-family: 'VFStudio', system-ui, sans-serif; font-size: ${fontSize}px; fill: ${escapeAttr(textColor)}; font-variation-settings: ${variationCss}; ${letterSpacingCss} }
   707→   707→</style>
   708→   708→</defs>
   709→   709→<rect x="0" y="0" width="${width}" height="${height}" fill="${escapeAttr(bgColor)}"/>
   710→   710→<text class="vftxt" text-anchor="${anchor}"${dirStyle ? ` style="${dirStyle}"` : ""}>${tspans}</text>
   711→   711→${overlaySvg}</svg>`;
   712→   712→}

   714→   714→// Loads an SVG string into an Image element and resolves only when the
   715→   715→// image is fully decoded. `decode()` waits for the SVG document's internal
   716→   716→// work (including font handling) to finish, which removes any chance of a
   717→   717→// half-rasterised frame (e.g. text missing on a few frames) being drawn.
   718→   718→async function loadSvgImage(
   719→   719→  svg: string,
   720→   720→  width: number,
   721→   721→  height: number
   722→   722→): Promise<HTMLImageElement> {
   723→   723→  const img = new Image(width, height);
   724→   724→  img.src = "data:image/svg+xml;charset=utf-8," + encodeURIComponent(svg);
   725→   725→  await new Promise<void>((resolve, reject) => {
   726→   726→    img.onload = () => resolve();
   727→   727→    img.onerror = () => reject(new Error("Failed to render SVG frame."));
   728→   728→  });
   729→   729→  if (typeof img.decode === "function") {
   730→   730→    try {
   731→   731→      await img.decode();
   732→   732→    } catch {
   733→   733→      // Already loaded via onload — safe to continue with what we have.
   734→   734→    }
   735→   735→  }
   736→   736→  return img;
   737→   737→}

   751→   751→// Compute the axis values for a normalised progress (0..1).
   752→   752→//
   753→   753→// mode = "sync": every animated axis shares one timeline — its value eases
   754→   754→// from animFrom to animTo as progress runs 0→1.
   755→   755→//
   756→   756→// mode = "binary": the animated axes count through a binary sequence. Each
   757→   757→// animated axis is one bit, ordered from the LEAST significant (the topmost
   758→   758→// animated axis) to the most significant (the bottom one). N animated axes
   759→   759→// cycle through 2^N states (e.g. 2 axes: 00 → 01 → 10 → 11 → 00). The whole
   760→   760→// timeline divides into 2^N equal segments; during segment k every axis eases
   761→   761→// from its value in state k to its value in state k+1 (the last segment wraps
   762→   762→// back to state 0, so the cycle always closes seamlessly at progress = 1).
   763→   763→// Bit 0 → animFrom, bit 1 → animTo of that axis. Pure function of progress,
   764→   764→// so preview and frame-by-frame export stay perfectly deterministic.
   765→   765→export function computeAxisValues(
   766→   766→  axes: {
   767→   767→    tag: string;
   768→   768→    min: number;
   769→   769→    max: number;
   770→   770→    animFrom: number;
   771→   771→    animTo: number;
   772→   772→    animate: boolean;
   773→   773→    current: number;
   774→   774→  }[],
   775→   775→  progress: number,
   776→   776→  easing: EasingType,
   777→   777→  mode: AxisMoveMode
   778→   778→): AxisValue[] {
   779→   779→  // Bit order: animated axes in display order (top → bottom); the topmost
   780→   780→  // animated axis is the least significant bit.
   781→   781→  const animIdx: number[] = [];
   782→   782→  axes.forEach((a, i) => {
   783→   783→    if (a.animate) animIdx.push(i);
   784→   784→  });
   785→   785→  const n = animIdx.length;
   786→   786→  const states = n > 0 ? 1 << n : 1;
   787→   787→  const p = clamp01(progress);
   788→   788→
   789→   789→  return axes.map((a, i) => {
   790→   790→    if (!a.animate) return { tag: a.tag, value: a.current };
   791→   791→    let value: number;
   792→   792→    if (mode === "binary" && n > 0) {
   793→   793→      const bit = animIdx.indexOf(i);
   794→   794→      const at = (s: number) => ((s >> bit) & 1 ? a.animTo : a.animFrom);
   795→   795→      const total = p * states;
   796→   796→      const seg = Math.min(states - 1, Math.floor(total));
   797→   797→      const local = clamp01(total - seg);
   798→   798→      const eased = applyEasing(local, easing);
   799→   799→      const from = at(seg);
   800→   800→      const to = at((seg + 1) % states);
   801→   801→      value = from + (to - from) * eased;
   802→   802→    } else {
   803→   803→      const eased = applyEasing(p, easing);
   804→   804→      value = a.animFrom + (a.animTo - a.animFrom) * eased;
   805→   805→    }
   806→   806→    return { tag: a.tag, value: clamp(value, a.min, a.max) };
   807→   807→  });
   808→   808→}
   809→   809→
   810→   810→export function clamp(n: number, min: number, max: number): number {
   811→   811→  return Math.min(max, Math.max(min, n));
   812→   812→}
