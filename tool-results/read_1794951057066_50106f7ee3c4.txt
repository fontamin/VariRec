     2→     2→// that embeds the variable font (base64) and applies `font-variation-settings`.
     3→     3→// SVG <text> supports variable-font axes through CSS, and keeping the font
     4→     4→// embedded as a data URI means the canvas never gets tainted, so we can call

     8→     8→// the axis name sits inline BEFORE the track (so toggling it never changes

   102→   102→export function hexToRgba(hex: string, alpha: number): string {
   103→   103→  let h = hex.replace("#", "");
   104→   104→  if (h.length === 3) {
   105→   105→    h = h
   106→   106→      .split("")
   107→   107→      .map((c) => c + c)
   108→   108→      .join("");
   109→   109→  }
   110→   110→  const r = parseInt(h.slice(0, 2), 16) || 0;
   111→   111→  const g = parseInt(h.slice(2, 4), 16) || 0;
   112→   112→  const b = parseInt(h.slice(4, 6), 16) || 0;
   113→   113→  return `rgba(${r},${g},${b},${alpha})`;
   114→   114→}

   118→   118→export function contrastFor(hex: string): string {
   119→   119→  let h = hex.replace("#", "");
   120→   120→  if (h.length === 3) {
   121→   121→    h = h
   122→   122→      .split("")
   123→   123→      .map((c) => c + c)
   124→   124→      .join("");
   125→   125→  }
   126→   126→  const r = parseInt(h.slice(0, 2), 16) || 0;
   127→   127→  const g = parseInt(h.slice(2, 4), 16) || 0;
   128→   128→  const b = parseInt(h.slice(4, 6), 16) || 0;
   129→   129→  const lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   130→   130→  return lum > 0.6 ? "#000000" : "#ffffff";
   131→   131→}

   179→   179→export function computeTextBlock(
   180→   180→  height: number,
   181→   181→  fontSize: number,
   182→   182→  lineHeight: number,
   183→   183→  lineCount: number,
   184→   184→  vAlign: VAlign
   185→   185→): TextBlock {
   186→   186→  const h = Math.max(1, lineCount) * fontSize * lineHeight;
   187→   187→  const margin = 24;
   188→   188→  if (vAlign === "top") return { top: margin, bottom: margin + h, height: h };
   189→   189→  if (vAlign === "bottom")
   190→   190→    return { top: height - margin - h, bottom: height - margin, height: h };
   191→   191→  const top = (height - h) / 2;
   192→   192→  return { top, bottom: top + h, height: h };
   193→   193→}

   235→   235→function cssCodeText(o: OverlayConfig): string {
   236→   236→  return `font-variation-settings: ${buildCssFontVariationSettings(
   237→   237→    o.axes.map((a) => ({ tag: a.tag, value: a.value }))
   238→   238→  )};`;

   241→   241→// The widest the CSS code string can get for the current axes. The rendered
   242→   242→// value strings vary in length from frame to frame — measuring against the
   243→   243→// widest variant keeps the CSS line's font size and bar geometry CONSTANT
   244→   244→// across frames (otherwise the bar visibly pulses while the animation runs).
   245→   245→function cssCodeWidestText(o: OverlayConfig): string {
   246→   246→  const dec = o.valueDecimals;
   247→   247→  const parts = o.axes.map((a) => {
   248→   248→    const widest = [a.min, a.max]
   249→   249→      .map((v) => formatValue(v, dec))
   250→   250→      .reduce((p, c) => (c.length > p.length ? c : p), "");
   251→   251→    return `"${a.tag}" ${widest}`;
   252→   252→  });
   253→   253→  return `font-variation-settings: ${parts.join(", ")};`;
   254→   254→}

   256→   256→function minMaxChars(o: OverlayConfig): number {
   257→   257→  let chars = 0;
   258→   258→  for (const a of o.axes) {
   259→   259→    const len = Math.max(fmtNum(a.min).length, fmtNum(a.max).length);
   260→   260→    if (len > chars) chars = len;
   261→   261→  }
   262→   262→  return Math.max(chars, 3);
   263→   263→}

   265→   265→// Shared geometry for the overlay — used by both the SVG frame renderer and
   266→   266→// the DOM live preview so both look identical. All values are in canvas px.
   267→   267→export function computeOverlayMetrics(
   268→   268→  width: number,
   269→   269→  height: number,
   270→   270→  o: OverlayConfig,
   271→   271→  textBlock: TextBlock
   272→   272→): OverlayMetrics {
   273→   273→  const s = o.scale;
   274→   274→  // The slider block only exists when the "slider" toggle is on; the name,
   275→   275→  // min–max labels and the value pill are decorations that live inside it.
   276→   276→  const slidersOn = o.showSliders && o.axes.length > 0;
   277→   277→  const fontPx = Math.max(9, Math.round(width * 0.016 * s));
   278→   278→  const pad = Math.max(8, Math.round(width * 0.014 * s));
   279→   279→  const innerPad = Math.max(6, Math.round(pad * 0.85));
   280→   280→  const panelRx = Math.max(6, Math.round(innerPad * 0.9));
   281→   281→  const trackH = Math.max(3, Math.round(fontPx * 0.3));
   282→   282→  const thumbR = Math.max(
   283→   283→    3,
   284→   284→    Math.round(fontPx * 0.48 * Math.min(3, Math.max(0.4, o.thumbScale)))
   285→   285→  );
   286→   286→  const thumbD = thumbR * 2;
   287→   287→  const minMaxFont = Math.max(8, Math.round(fontPx * 0.78));
   288→   288→
   289→   289→  // Inline axis name — drawn BEFORE the track on the same row, so toggling
   290→   290→  // the name never changes the height of the slider stack.
   291→   291→  const hasName = slidersOn && (o.showName || o.showTag) && o.axes.length > 0;
   292→   292→  let nameChars = 0;
   293→   293→  if (hasName) {
   294→   294→    for (const a of o.axes) {
   295→   295→      let len = 0;
   296→   296→      if (o.showName) len += a.name.length;
   297→   297→      if (o.showTag) len += a.tag.length + 2;
   298→   298→      if (len > nameChars) nameChars = len;
   299→   299→    }
   300→   300→  }
   301→   301→  const nameW = hasName ? Math.ceil(nameChars * 0.62 * fontPx) : 0;
   302→   302→  // Horizontal gap between the inline name and the min label / track.
   303→   303→  const nameGapPx = Math.round(
   304→   304→    Math.min(Math.max(o.nameGap, 0), 4) * fontPx
   305→   305→  );
   306→   306→  const textGapPx = Math.round(
   307→   307→    Math.min(Math.max(o.textGap, 0), 6) * fontPx
   308→   308→  );
   309→   309→
   310→   310→  // Value pill riding ON the thumb (Material-style label). It replaces the
   311→   311→  // thumb and adds NO extra row, so the stack height stays identical whether
   312→   312→  // the value number is on or off. Width uses the widest value string the
   313→   313→  // axis can ever show (min/max at the chosen decimals) so the pill geometry
   314→   314→  // is frame-stable — no pulsing mid-animation.
   315→   315→  const pillOn = slidersOn && o.showLiveValue;
   316→   316→  const pillFont = Math.max(8, Math.round(fontPx * 0.8));
   317→   317→  const pillPadX = Math.max(6, Math.round(pillFont * 0.55));
   318→   318→  const pillPadY = Math.max(3, Math.round(pillFont * 0.3));
   319→   319→  let valueChars = 3;
   320→   320→  for (const a of o.axes) {
   321→   321→    const len = Math.max(
   322→   322→      formatValue(a.min, o.valueDecimals).length,
   323→   323→      formatValue(a.max, o.valueDecimals).length
   324→   324→    );
   325→   325→    if (len > valueChars) valueChars = len;
   326→   326→  }
   327→   327→  const pillTextW = Math.ceil(valueChars * 0.62 * pillFont);
   328→   328→  const pillH = Math.max(thumbD, Math.round(pillFont * 1.5) + pillPadY * 2);
   329→   329→  const pillW = Math.max(
   330→   330→    thumbD + Math.round(thumbD * 0.4),
   331→   331→    pillTextW + pillPadX * 2
   332→   332→  );
   333→   333→  const pillTextFill = contrastFor(o.color);
   334→   334→
   335→   335→  const minMaxLabelW =
   336→   336→    slidersOn && o.showMinMax
   337→   337→      ? Math.ceil(minMaxChars(o) * 0.62 * minMaxFont)
   338→   338→      : 0;
   339→   339→  const flankGap = Math.max(4, Math.round(fontPx * 0.5));
   340→   340→  const trackRowH = Math.max(
   341→   341→    Math.round(thumbD * 1.25),
   342→   342→    Math.round(minMaxFont * 1.35)
   343→   343→  );
   344→   344→
   345→   345→  const nameSpan = nameW + (hasName ? nameGapPx : 0);
   346→   346→  const trackFlank = minMaxLabelW > 0 ? minMaxLabelW * 2 + flankGap * 2 : 0;
   347→   347→  const availForTrack =
   348→   348→    width - pad * 2 - innerPad * 2 - nameSpan - trackFlank;
   349→   349→  const trackW = Math.max(
   350→   350→    fontPx * 6,
   351→   351→    Math.min(
   352→   352→      Math.round(width * 0.3 * s),
   353→   353→      Math.max(availForTrack, fontPx * 6)
   354→   354→    )
   355→   355→  );
   356→   356→
   357→   357→  const trackRowW = nameSpan + trackFlank + trackW;
   358→   358→  const maxBlock = Math.max(fontPx * 6, width - pad * 2 - innerPad * 2);
   359→   359→  const blockW = Math.min(maxBlock, trackRowW);
   360→   360→  const panelW = blockW + innerPad * 2;
   361→   361→
   362→   362→  const axisGap = Math.round(fontPx * 1.2);
   363→   363→  const rowStack = trackRowH; // one compact row per axis
   364→   364→  const n = o.axes.length;
   365→   365→  const panelH =
   366→   366→    slidersOn && n > 0 ? innerPad * 2 + rowStack * n + axisGap * (n - 1) : 0;
   367→   367→
   368→   368→  // CSS code line — auto-shrink font if the string is longer than the bar.
   369→   369→  // The width estimate uses the widest possible value strings so the font
   370→   370→  // size (and therefore the bar geometry) stays constant across frames.
   371→   371→  const cssText = cssCodeText(o);
   372→   372→  const cssWidest = cssCodeWidestText(o);
   373→   373→  const availCss = Math.max(40, width - pad * 2 - innerPad * 2);
   374→   374→  const cssEst = 0.62 * fontPx * Math.max(1, cssWidest.length);
   375→   375→  const cssFontPx =
   376→   376→    cssEst > availCss
   377→   377→      ? Math.max(
   378→   378→          8,
   379→   379→          Math.floor(availCss / (0.62 * Math.max(1, cssWidest.length)))
   380→   380→        )
   381→   381→      : fontPx;
   382→   382→  const cssTextW = Math.ceil(cssWidest.length * 0.62 * cssFontPx);
   383→   383→  const cssBarH = o.showCss ? Math.round(cssFontPx * 1.5 + innerPad * 1.1) : 0;
   384→   384→  const cssGapPx = Math.round(
   385→   385→    Math.min(Math.max(o.cssGap, 0), 4) * fontPx
   386→   386→  );
   387→   387→  const cssOn = o.showCss && cssBarH > 0;
   388→   388→  const cssBarW = cssOn
   389→   389→    ? Math.min(
   390→   390→        width - pad * 2,
   391→   391→        o.centered ? cssTextW + innerPad * 2 : width - pad * 2
   392→   392→      )
   393→   393→    : 0;
   394→   394→
   395→   395→  // ── Vertical placement ────────────────────────────────────────────────
   396→   396→  // The whole stack (slider panel + CSS line) hugs the text block with
   397→   397→  // textGapPx between them, clamped so it never leaves the canvas.
   398→   398→  const stackH = panelH + (slidersOn && cssOn ? cssGapPx + cssBarH : cssOn ? cssBarH : 0);
   399→   399→  const minTop = pad;
   400→   400→  const maxTop = Math.max(minTop, height - pad - stackH);
   401→   401→  let panelY = 0;
   402→   402→  if (o.position === "top") {
   403→   403→    panelY = Math.round(
   404→   404→      Math.min(maxTop, Math.max(minTop, textBlock.top - textGapPx - stackH))
   405→   405→    );
   406→   406→  } else {
   407→   407→    panelY = Math.round(
   408→   408→      Math.min(maxTop, Math.max(minTop, textBlock.bottom + textGapPx))
   409→   409→    );
   410→   410→  }
   411→   411→  const cssBarY = cssOn
   412→   412→    ? Math.round(
   413→   413→        Math.min(
   414→   414→          height - pad - cssBarH,
   415→   415→          Math.max(
   416→   416→            pad,
   417→   417→            slidersOn ? panelY + panelH + cssGapPx : panelY
   418→   418→          )
   419→   419→        )
   420→   420→      )
   421→   421→    : 0;
   422→   422→
   423→   423→  const panelX = o.centered
   424→   424→    ? Math.max(pad, Math.round((width - panelW) / 2))
   425→   425→    : pad;
   426→   426→  const cssBarX = cssOn
   427→   427→    ? o.centered
   428→   428→      ? Math.max(pad, Math.round((width - cssBarW) / 2))
   429→   429→      : pad
   430→   430→    : 0;
   431→   431→
   432→   432→  const trackLeft = nameSpan + (minMaxLabelW > 0 ? minMaxLabelW + flankGap : 0);
   433→   433→
   434→   434→  return {
   435→   435→    pad,
   436→   436→    innerPad,
   437→   437→    panelRx,
   438→   438→    fontPx,
   439→   439→    minMaxFont,
   440→   440→    nameW,
   441→   441→    nameGapPx,
   442→   442→    textGapPx,

   475→   475→// Thumb marker in every supported shape, drawn centered on (cx, cy).
   476→   476→function thumbSvg(
   477→   477→  shape: ThumbShape,
   478→   478→  cx: number,
   479→   479→  cy: number,
   480→   480→  r: number,
   481→   481→  fill: string
   482→   482→): string {
   483→   483→  const d = r * 2;
   484→   484→  const x = round2(cx - r);
   485→   485→  const y = round2(cy - r);
   486→   486→  switch (shape) {
   487→   487→    case "square":
   488→   488→      return `<rect x="${x}" y="${y}" width="${d}" height="${d}" fill="${fill}"/>`;
   489→   489→    case "rounded":
   490→   490→      return `<rect x="${x}" y="${y}" width="${d}" height="${d}" rx="${round2(r * 0.42)}" fill="${fill}"/>`;
   491→   491→    case "diamond":
   492→   492→      return `<rect x="${x}" y="${y}" width="${d}" height="${d}" rx="${round2(r * 0.18)}" fill="${fill}" transform="rotate(45 ${round2(cx)} ${round2(cy)})"/>`;
   493→   493→    default:
   494→   494→      return `<circle cx="${round2(cx)}" cy="${round2(cy)}" r="${r}" fill="${fill}"/>`;
   495→   495→  }

   498→   498→function buildOverlaySvg(
   499→   499→  o: OverlayConfig,
   500→   500→  width: number,
   501→   501→  height: number,
   502→   502→  textBlock: TextBlock
   503→   503→): string {
   504→   504→  const m = computeOverlayMetrics(width, height, o, textBlock);
   505→   505→  const parts: string[] = [];
   506→   506→  const fontAttr = `font-family="${escapeAttr(o.font)}"`;
   507→   507→  const fill = escapeAttr(o.color);
   508→   508→  if (o.showSliders && m.panelH > 0) {
   509→   509→    if (o.bgOpacity > 0) {
   510→   510→      parts.push(
   511→   511→        `<rect x="${m.panelX}" y="${m.panelY}" width="${m.panelW}" height="${m.panelH}" rx="${m.panelRx}" fill="${escapeAttr(o.bgColor)}" fill-opacity="${round2(o.bgOpacity)}"/>`
   512→   512→      );
   513→   513→    }
   514→   514→    const contentX = m.panelX + m.innerPad;
   515→   515→
   516→   516→    o.axes.forEach((a, i) => {
   517→   517→      const y = m.panelY + m.innerPad + i * (m.trackRowH + m.axisGap);
   518→   518→      const cy = y + m.trackRowH / 2;
   519→   519→      const t = clamp01((a.value - a.min) / (a.max - a.min || 1));
   520→   520→      const trackX = contentX + m.trackLeft;
   521→   521→      const fillW = t * m.trackW;
   522→   522→      const thumbCx = trackX + fillW;
   523→   523→
   524→   524→      // Inline axis name, BEFORE the track (same row → constant stack height).
   525→   525→      // Right-aligned inside its fixed-width slot, so every name ends at the
   526→   526→      // SAME distance from the start of its own track.
   527→   527→      if (m.nameW > 0) {
   528→   528→        const label = [
   529→   529→          o.showName ? escapeXml(a.name) : "",
   530→   530→          o.showTag ? `(${escapeXml(a.tag)})` : "",
   531→   531→        ]
   532→   532→          .filter(Boolean)
   533→   533→          .join(" ");
   534→   534→        parts.push(
   535→   535→          `<text x="${round2(contentX + m.nameW)}" y="${round2(cy + m.fontPx * 0.35)}" text-anchor="end" fill="${fill}" ${fontAttr} font-size="${m.fontPx}">${label}</text>`
   536→   536→        );
   537→   537→      }
   538→   538→
   539→   539→      // Min–max labels flanking the track.
   540→   540→      if (m.minMaxLabelW > 0) {
   541→   541→        const lb = cy + m.minMaxFont * 0.35;
   542→   542→        parts.push(
   543→   543→          `<text x="${round2(trackX - m.flankGap)}" y="${round2(lb)}" text-anchor="end" fill="${fill}" fill-opacity="0.75" ${fontAttr} font-size="${m.minMaxFont}">${fmtNum(a.min)}</text>`
   544→   544→        );
   545→   545→        parts.push(
   546→   546→          `<text x="${round2(trackX + m.trackW + m.flankGap)}" y="${round2(lb)}" text-anchor="start" fill="${fill}" fill-opacity="0.75" ${fontAttr} font-size="${m.minMaxFont}">${fmtNum(a.max)}</text>`
   547→   547→        );
   548→   548→      }
   549→   549→
   550→   550→      parts.push(
   551→   551→        `<rect x="${trackX}" y="${round2(cy - m.trackH / 2)}" width="${m.trackW}" height="${m.trackH}" rx="${m.trackH / 2}" fill="${fill}" fill-opacity="0.22"/>`
   552→   552→      );
   553→   553→      if (fillW > 0) {
   554→   554→        parts.push(
   555→   555→          `<rect x="${trackX}" y="${round2(cy - m.trackH / 2)}" width="${round2(fillW)}" height="${m.trackH}" rx="${m.trackH / 2}" fill="${fill}" fill-opacity="0.7"/>`
   556→   556→        );
   557→   557→      }
   558→   558→
   559→   559→      // Value pill riding ON the thumb (replaces it), or the plain thumb.
   560→   560→      // The pill centre is snapped to whole canvas pixels so it rasterises
   561→   561→      // identically frame-to-frame instead of shimmering sub-pixel.
   562→   562→      if (m.pillOn) {
   563→   563→        const pcx = Math.round(
   564→   564→          Math.min(
   565→   565→            trackX + m.trackW - m.pillW / 2,
   566→   566→            Math.max(trackX + m.pillW / 2, thumbCx)
   567→   567→          )
   568→   568→        );
   569→   569→        const prx = pillRadius(o.thumbShape, Math.min(m.pillW, m.pillH));
   570→   570→        parts.push(
   571→   571→          `<rect x="${round2(pcx - m.pillW / 2)}" y="${round2(cy - m.pillH / 2)}" width="${round2(m.pillW)}" height="${round2(m.pillH)}" rx="${round2(prx)}" fill="${fill}"/>`
   572→   572→        );
   573→   573→        parts.push(
   574→   574→          `<text x="${pcx}" y="${round2(cy + m.pillFont * 0.35)}" text-anchor="middle" fill="${m.pillTextFill}" ${fontAttr} font-size="${m.pillFont}">${formatValue(a.value, o.valueDecimals)}</text>`
   575→   575→        );
   576→   576→      } else {
   577→   577→        parts.push(thumbSvg(o.thumbShape, thumbCx, cy, m.thumbR, fill));
   578→   578→      }
   579→   579→    });
   580→   580→  }
   581→   581→
   582→   582→  if (m.cssOn) {
   583→   583→    const cssText = cssCodeText(o);
   584→   584→    if (o.bgOpacity > 0) {
   585→   585→      parts.push(
   586→   586→        `<rect x="${m.cssBarX}" y="${m.cssBarY}" width="${m.cssBarW}" height="${m.cssBarH}" rx="${m.panelRx}" fill="${escapeAttr(o.bgColor)}" fill-opacity="${round2(o.bgOpacity)}"/>`
   587→   587→      );
   588→   588→    }
   589→   589→    const anchor = o.centered ? "middle" : "start";
   590→   590→    const tx = o.centered
   591→   591→      ? m.cssBarX + m.cssBarW / 2
   592→   592→      : m.cssBarX + m.innerPad;
   593→   593→    parts.push(
   594→   594→      `<text x="${round2(tx)}" y="${round2(m.cssBarY + m.cssBarH / 2 + m.cssFontPx * 0.36)}" text-anchor="${anchor}" fill="${fill}" ${fontAttr} font-size="${m.cssFontPx}" xml:space="preserve">${escapeXml(cssText)}</text>`
   595→   595→    );
   596→   596→  }
   597→   597→
   598→   598→  return parts.join("");
   599→   599→}

   601→   601→export function buildSvgString(cfg: RenderConfig): string {
   602→   602→  const {
   603→   603→    text,
   604→   604→    fontBase64,
   605→   605→    fontCssFormat,
   606→   606→    fontMime,
   607→   607→    variation,
   608→   608→    fontSize,
   609→   609→    textColor,
   610→   610→    bgColor,
   611→   611→    width,
   612→   612→    height,
   613→   613→    letterSpacing,
   614→   614→    lineHeight,
   615→   615→    align,
   616→   616→    vAlign,
   617→   617→    overlay,
   618→   618→  } = cfg;
   619→   619→
   620→   620→  const lines = text.length ? text.split("\n") : [" "];
   621→   621→  const textBlock = computeTextBlock(
   622→   622→    height,
   623→   623→    fontSize,
   624→   624→    lineHeight,
   625→   625→    lines.length,
   626→   626→    vAlign
   627→   627→  );
   628→   628→  const top = textBlock.top;
   629→   629→  const anchorX =
   630→   630→    align === "center" ? width / 2 : align === "right" ? width - 24 : 24;
   631→   631→  const anchor =
   632→   632→    align === "center" ? "middle" : align === "right" ? "end" : "start";
   633→   633→
   634→   634→  const tspans = lines
   635→   635→    .map((line, i) => {
   636→   636→      const y = top + i * fontSize * lineHeight + fontSize * 0.78;
   637→   637→      return `<tspan x="${anchorX}" y="${round2(y)}">${escapeXml(line) || "&#8203;"}</tspan>`;
   638→   638→    })
   639→   639→    .join("");
   640→   640→
   641→   641→  const variationCss = buildVariationString(variation);
   642→   642→  const letterSpacingCss =
   643→   643→    letterSpacing !== 0 ? `letter-spacing: ${letterSpacing}px;` : "";
   644→   644→
   645→   645→  const overlaySvg =
   646→   646→    overlay &&
   647→   647→    ((overlay.showSliders && overlay.axes.length > 0) || overlay.showCss)
   648→   648→      ? buildOverlaySvg(overlay, width, height, textBlock)
   649→   649→      : "";
   650→   650→
   651→   651→  return `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}" viewBox="0 0 ${width} ${height}">
   652→   652→<defs>
   653→   653→<style>
   654→   654→@font-face { font-family: 'VFStudio'; src: url(data:${escapeAttr(fontMime)};base64,${fontBase64}) format('${escapeAttr(fontCssFormat)}'); }
   655→   655→.vftxt { font-family: 'VFStudio', system-ui, sans-serif; font-size: ${fontSize}px; fill: ${escapeAttr(textColor)}; font-variation-settings: ${variationCss}; ${letterSpacingCss} }
   656→   656→</style>
   657→   657→</defs>
   658→   658→<rect x="0" y="0" width="${width}" height="${height}" fill="${escapeAttr(bgColor)}"/>
   659→   659→<text class="vftxt" text-anchor="${anchor}">${tspans}</text>
   660→   660→${overlaySvg}</svg>`;
   661→   661→}

   663→   663→// Loads an SVG string into an Image element and resolves only when the
   664→   664→// image is fully decoded. `decode()` waits for the SVG document's internal
   665→   665→// work (including font handling) to finish, which removes any chance of a
   666→   666→// half-rasterised frame (e.g. text missing on a few frames) being drawn.
   667→   667→async function loadSvgImage(
   668→   668→  svg: string,
   669→   669→  width: number,
   670→   670→  height: number
   671→   671→): Promise<HTMLImageElement> {
   672→   672→  const img = new Image(width, height);
   673→   673→  img.src = "data:image/svg+xml;charset=utf-8," + encodeURIComponent(svg);
   674→   674→  await new Promise<void>((resolve, reject) => {
   675→   675→    img.onload = () => resolve();
   676→   676→    img.onerror = () => reject(new Error("Failed to render SVG frame."));
   677→   677→  });
   678→   678→  if (typeof img.decode === "function") {
   679→   679→    try {
   680→   680→      await img.decode();
   681→   681→    } catch {
   682→   682→      // Already loaded via onload — safe to continue with what we have.
   683→   683→    }
   684→   684→  }
   685→   685→  return img;
   686→   686→}

   700→   700→// Compute the axis values for a normalised progress (0..1).
   701→   701→//
   702→   702→// mode = "sync": every animated axis shares one timeline — its value eases
   703→   703→// from animFrom to animTo as progress runs 0→1.
   704→   704→//
   705→   705→// mode = "binary": the animated axes count through a binary sequence. Each
   706→   706→// animated axis is one bit, ordered from the LEAST significant (the topmost
   707→   707→// animated axis) to the most significant (the bottom one). N animated axes
   708→   708→// cycle through 2^N states (e.g. 2 axes: 00 → 01 → 10 → 11 → 00). The whole
   709→   709→// timeline divides into 2^N equal segments; during segment k every axis eases
   710→   710→// from its value in state k to its value in state k+1 (the last segment wraps
   711→   711→// back to state 0, so the cycle always closes seamlessly at progress = 1).
   712→   712→// Bit 0 → animFrom, bit 1 → animTo of that axis. Pure function of progress,
   713→   713→// so preview and frame-by-frame export stay perfectly deterministic.
   714→   714→export function computeAxisValues(
   715→   715→  axes: {
   716→   716→    tag: string;
   717→   717→    min: number;
   718→   718→    max: number;
   719→   719→    animFrom: number;
   720→   720→    animTo: number;
   721→   721→    animate: boolean;
   722→   722→    current: number;
   723→   723→  }[],
   724→   724→  progress: number,
   725→   725→  easing: EasingType,
   726→   726→  mode: AxisMoveMode
   727→   727→): AxisValue[] {
   728→   728→  // Bit order: animated axes in display order (top → bottom); the topmost
   729→   729→  // animated axis is the least significant bit.
   730→   730→  const animIdx: number[] = [];
   731→   731→  axes.forEach((a, i) => {
   732→   732→    if (a.animate) animIdx.push(i);
   733→   733→  });
   734→   734→  const n = animIdx.length;
   735→   735→  const states = n > 0 ? 1 << n : 1;
   736→   736→  const p = clamp01(progress);
   737→   737→
   738→   738→  return axes.map((a, i) => {
   739→   739→    if (!a.animate) return { tag: a.tag, value: a.current };
   740→   740→    let value: number;
   741→   741→    if (mode === "binary" && n > 0) {
   742→   742→      const bit = animIdx.indexOf(i);
   743→   743→      const at = (s: number) => ((s >> bit) & 1 ? a.animTo : a.animFrom);
   744→   744→      const total = p * states;
   745→   745→      const seg = Math.min(states - 1, Math.floor(total));
   746→   746→      const local = clamp01(total - seg);
   747→   747→      const eased = applyEasing(local, easing);
   748→   748→      const from = at(seg);
   749→   749→      const to = at((seg + 1) % states);
   750→   750→      value = from + (to - from) * eased;
   751→   751→    } else {
   752→   752→      const eased = applyEasing(p, easing);
   753→   753→      value = a.animFrom + (a.animTo - a.animFrom) * eased;
   754→   754→    }
   755→   755→    return { tag: a.tag, value: clamp(value, a.min, a.max) };
   756→   756→  });
   757→   757→}
   758→   758→
   759→   759→export function clamp(n: number, min: number, max: number): number {
   760→   760→  return Math.min(max, Math.max(min, n));
   761→   761→}
