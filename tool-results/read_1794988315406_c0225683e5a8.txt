   574→}) {
   575→  return (
   576→    <Popover>
   577→      <PopoverTrigger asChild>

   708→function ShapeGlyph({ shape }: { shape: ThumbShape }) {
   709→  const base = "block size-3 bg-foreground";
   710→  switch (shape) {
   711→    case "square":
   712→      return <span className={base} />;
   713→    case "rounded":
   714→      return <span className={cn(base, "rounded-[4px]")} />;
   715→    case "diamond":
   716→      return <span className={cn(base, "rotate-45 rounded-[2px]")} />;
   717→    default:
   718→      return <span className={cn(base, "rounded-full")} />;
   719→  }
